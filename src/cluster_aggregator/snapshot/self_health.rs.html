14/// A point-in-time view of the health of Checkmk Kubernetes Agent and its
15/// components: cluster-aggregator and node-scraper.
16///
17/// This is included in the [`crate::snapshot::Snapshot`] and used to generate
18/// cluster-level sections for alerting about the state of the agent, such as
19/// when a node last reported in (node-scraper), or if the reflectors had
20/// errors.
21///
22/// Scheduled Pods belonging to the node-scraper DaemonSet are the source of
23/// truth for which nodes should report. This accounts for nodes on which the
24/// DaemonSet is intentionally not scheduled while still retaining missing
25/// cache entries for broken node-scraper Pods.
26#[derive(Debug, Default)]
27pub struct SelfHealth {
28    /// Health data for each of the payloads sent by node-scraper, keyed by
29    /// node name.
30    pub node_node_scrapers: BTreeMap<String, NodeNodeScraperHealth>,
31    /// Maps kind names to reflector states.
32    pub(crate) reflector_healths: BTreeMap<&'static str, ReflectorHealth>,
33    /// Build information about cluster-aggregator.
34    pub cluster_aggregator_build_info: ClusterAggregatorBuildInfo,
35}

77impl ReflectorHealth {
78    fn from_reflector_health(health: reflectors::ReflectorHealth, now: Instant) -> Self {
79        Self {
80            has_been_initialized: health.has_been_initialized,
81            relist_started_age: health
82                .relist_started_at
83                .map(|i| now.saturating_duration_since(i)),
84            relist_completed_age: health
85                .relist_completed_at
86                .map(|i| now.saturating_duration_since(i)),
87            relist_duration: health.relist_duration,
88            last_error_age: health
89                .last_error_at
90                .map(|i| now.saturating_duration_since(i)),
91            errors_total: health.errors_total,
92        }
93    }
94}

96impl SelfHealth {
97    pub(crate) fn new(
98        now: Instant,
99        expected_nodes: &BTreeSet<String>,
100        reflector_healths: FrozenReflectorHealths,
101        kubelet_stats_summary_cache: &Cache<String, Arc<NodeScraperIngestion<StatsSummary>>>,
102        kubelet_health_cache: &Cache<String, Arc<NodeScraperIngestion<KubeletHealth>>>,
103        system_agent_cache: &Cache<String, Arc<NodeScraperIngestion<SystemAgentOutput>>>,
104    ) -> SelfHealth {
105        let kubelet_stats = Self::cache_health(now, expected_nodes, kubelet_stats_summary_cache);
106        let mut kubelet_health = Self::cache_health(now, expected_nodes, kubelet_health_cache);
107        let mut system_agent = Self::cache_health(now, expected_nodes, system_agent_cache);
108        let node_node_scrapers = kubelet_stats
109            .into_iter()
110            .map(|(node, kubelet_stats)| {
111                let kubelet_health = kubelet_health.remove(&node).unwrap_or_default();
112                let system_agent = system_agent.remove(&node).unwrap_or_default();
113                (
114                    node,
115                    NodeNodeScraperHealth {
116                        kubelet_stats,
117                        kubelet_health,
118                        system_agent,
119                    },
120                )
121            })
122            .collect();
123        let reflector_healths = Self::reflector_healths_from_frozen(now, reflector_healths);
124        let cluster_aggregator_build_info = ClusterAggregatorBuildInfo {
125            version: env!("CARGO_PKG_VERSION"),
126            git_sha: option_env!("CHECKMK_KUBERNETES_AGENT_GIT_SHA"),
127        };
128        SelfHealth {
129            node_node_scrapers,
130            reflector_healths,
131            cluster_aggregator_build_info,
132        }
133    }

135    pub(crate) fn expected_node_names(
136        daemonsets: &[Arc<DaemonSet>],
137        owner_graph: &OwnerGraph,
138        identity: Option<&NodeScraperDaemonSet>,
139    ) -> BTreeSet<String> {
140        let Some(identity) = identity else {
141            return BTreeSet::new();
142        };
143        let Some(uid) = daemonsets
144            .iter()
145            .find(|daemonset| {
146                daemonset.metadata.namespace.as_deref() == Some(&identity.namespace)
147                    && daemonset.metadata.name.as_deref() == Some(&identity.name)
148            })
149            .and_then(|daemonset| daemonset.metadata.uid.as_deref())
150        else {
151            return BTreeSet::new();
152        };
153
154        owner_graph
155            .pods_by_controller(uid)
156            .iter()
157            .filter_map(|pod| pod.spec.as_ref()?.node_name.clone())
158            .collect()
159    }

161    /// Given an `Instant` representing _the moment the `Snapshot` is being
162    /// taken_, a collection of expected node names, and a cache of
163    /// node-scraper ingestions, calculate how long it has been since each
164    /// node reported in and collect the results, keyed on the node name.
165    ///
166    /// The node names are taken as the source of truth for "which nodes are
167    /// supposed to be present?" Anything that is not in the cache but is in
168    /// this collection is returned with empty health data, with
169    /// the expectation that monitoring reports the node as having stopped
170    /// reporting its metrics.
171    fn cache_health<T>(
172        now: Instant,
173        nodes: &BTreeSet<String>,
174        cache: impl IntoIterator<Item = (Arc<String>, Arc<NodeScraperIngestion<T>>)>,
175    ) -> BTreeMap<String, NodeScraperIngestionHealth> {
176        let mut intermediary: HashMap<String, NodeScraperIngestionHealth> = cache
177            .into_iter()
178            .map(|(name, ingestion)| {
179                (
180                    name.to_string(),
181                    NodeScraperIngestionHealth {
182                        last_heard_age: Some(now.saturating_duration_since(ingestion.received_at)),
183                        scrape_time: ingestion.metadata.scrape_time,
184                        version: ingestion.metadata.version.clone(),
185                        git_sha: ingestion.metadata.git_sha.clone(),
186                    },
187                )
188            })
189            .collect();
190
191        let mut map = BTreeMap::new();
192        for node_name in nodes {
193            let health = intermediary.remove(node_name).unwrap_or_default();
194            map.insert(node_name.clone(), health);
195        }
196        map
197    }
198
199    /// Given an `Instant` representing _the moment the `Snapshot` is being
200    /// taken_, and a [`crate::ingest::reflectors::FrozenReflectorHealths`],
201    /// generate a `BTreeMap` using the `IntoIterator` instance of the
202    /// `FrozenReflectorHealths`, converting each reflector health into a
203    /// [`crate::snapshot::self_health::ReflectorHealth`].
204    fn reflector_healths_from_frozen(
205        now: Instant,
206        healths: FrozenReflectorHealths,
207    ) -> BTreeMap<&'static str, ReflectorHealth> {
208        healths
209            .into_iter()
210            .map(|(kind, health)| (kind, ReflectorHealth::from_reflector_health(health, now)))
211            .collect()
212    }

231    #[test]
232    fn cache_health() {
233        let now = Instant::now();
234        let nodes =
235            BTreeSet::from(["node1334", "node1335", "node1336", "node1337"].map(str::to_string));
236        let cache = [
237            (
238                s("node1334").into(),
239                ingestion(now - Duration::from_mins(5)),
240            ),
241            (
242                s("node1335").into(),
243                ingestion(now - Duration::from_mins(3)),
244            ),
245            (
246                s("node1336").into(),
247                ingestion(now - Duration::from_secs(26)),
248            ),
249            (
250                s("decommissioned01").into(),
251                ingestion(now - Duration::from_mins(29)),
252            ),
253        ];
254        let healths = SelfHealth::cache_health(now, &nodes, cache);
255
256        assert_eq!(healths.len(), nodes.len());
257        assert_eq!(
258            healths["node1334"].last_heard_age,
259            Some(Duration::from_mins(5))
260        );
261        assert_eq!(
262            healths["node1335"].last_heard_age,
263            Some(Duration::from_mins(3))
264        );
265        assert_eq!(
266            healths["node1336"].last_heard_age,
267            Some(Duration::from_secs(26))
268        );
269
270        // Expected node not in cache gets added to the result as None
271        assert!(healths["node1337"].last_heard_age.is_none());
272
273        // A node in the cache but not in the expected set is not added
274        assert!(!healths.contains_key("decommissioned01"));
275    }
